def analyse(title, summary=""):
    text = (title + " " + (summary or "")).strip()
    score, reasons = 0, []
    used = set()
    for pat, w, label in POS_C + NEG_C:
        m = pat.search(text)
        if m and label not in used:
            used.add(label)
            score += w
            reasons.append({"word": m.group(0).strip()[:40], "label": label, "weight": w})
    score = max(-8, min(8, score))
    sent = "pos" if score >= 1 else "neg" if score <= -1 else "neu"
    cat = "General"
    for name, pat in CATS:
        if pat.search(text):
            cat = name
            break
    return sent, score, reasons, cat

def http_get(url, timeout=12, headers=None):
    req = urllib.request.Request(url, headers={"User-Agent": UA, "Accept": "*/*",
                                               "Accept-Language": "en-US,en;q=0.9", **(headers or {})})
    with _opener.open(req, timeout=timeout) as r:
        return r.read()


def get_crumb(force=False):
    with LOCK:
        if _crumb["v"] and not force and time.time() - _crumb["t"] < 3600:
            return _crumb["v"]
        try:
            http_get("https://fc.yahoo.com", timeout=8)
        except Exception:
            pass  # this endpoint returns 404 but still sets the cookie
        for host in ("query1", "query2"):
            try:
                c = http_get(f"https://{host}.finance.yahoo.com/v1/test/getcrumb", timeout=8).decode().strip()
                if c and "<" not in c:
                    _crumb.update(v=c, t=time.time())
                    return c
            except Exception:
                continue
    return None


def yahoo_json(path, params, retry=True):
    last = None
    for host in ("query1", "query2"):
        url = f"https://{host}.finance.yahoo.com{path}?" + urllib.parse.urlencode(params)
        try:
            return json.loads(http_get(url))
        except urllib.error.HTTPError as e:
            last = e
            if e.code in (401, 403, 429) and retry:
                break
        except Exception as e:
            last = e
    raise last or RuntimeError("Yahoo request failed")

def flatten_fund(qs):
    out = {}
    for mod in ("summaryDetail", "defaultKeyStatistics", "financialData", "majorHoldersBreakdown", "price"):
        for k, v in (qs.get(mod) or {}).items():
            val = raw(v) if isinstance(v, dict) else v
            if isinstance(val, (int, float, str)) and k not in out:
                out[k] = val
    return out


def parse_earnings(qs):
    hist = []
    for h in (qs.get("earningsHistory") or {}).get("history", []):
        act, est = raw(h.get("epsActual")), raw(h.get("epsEstimate"))
        if act is None:
            continue
        sp = None
        if est not in (None, 0):
            sp = (act - est) / abs(est) * 100
        hist.append({"quarter": raw(h.get("quarter")), "epsActual": act, "epsEstimate": est, "surprisePct": sp})
    hist.sort(key=lambda x: x["quarter"] or 0)
    fc = (qs.get("earnings") or {}).get("financialsChart") or {}

    def fin(rows):
        return [{"date": r.get("date"), "revenue": raw(r.get("revenue")), "earnings": raw(r.get("earnings"))}
                for r in rows or []]
    nxt = None
    ed = ((qs.get("calendarEvents") or {}).get("earnings") or {}).get("earningsDate") or []
    if ed:
        nxt = raw(ed[0])
    trend = {}
    for t in (qs.get("recommendationTrend") or {}).get("trend", []):
        if t.get("period") == "0m":
            trend = {k: t.get(k, 0) for k in ("strongBuy", "buy", "hold", "sell", "strongSell")}
    return {"history": hist, "quarterly": fin(fc.get("quarterly")), "yearly": fin(fc.get("yearly")),
            "nextDate": nxt, "recs": trend}

def clean_html(s):
    s = re.sub(r"<[^>]+>", " ", s or "")
    return re.sub(r"\s+", " ", html.unescape(s)).strip()


def fetch_google_news(query):
    q = urllib.parse.quote(query + " when:14d")
    url = f"https://news.google.com/rss/search?q={q}&hl=en-IN&gl=IN&ceid=IN:en"
    root = ET.fromstring(http_get(url, timeout=10))
    items = []
    for it in root.iter("item"):
        title = (it.findtext("title") or "").strip()
        src = (it.findtext("source") or "").strip()
        if src and title.endswith(" - " + src):
            title = title[: -len(src) - 3].strip()
        try:
            ts = int(parsedate_to_datetime(it.findtext("pubDate")).timestamp())
        except Exception:
            ts = 0
        desc = clean_html(it.findtext("description"))
        if desc.startswith(title):
            desc = desc[len(title):].strip(" - ")
        if src and desc.endswith(src):
            desc = desc[: -len(src)].strip()
        items.append({"title": title, "source": src or "Google News", "link": it.findtext("link") or "",
                      "ts": ts, "summary": desc if len(desc) > 25 else ""})
    return items

def meta_description(url):
    try:
        page = http_get(url, timeout=5).decode("utf-8", "ignore")[:200000]
    except Exception:
        return ""
    for pat in (r'<meta[^>]+property=["\']og:description["\'][^>]+content=["\']([^"\']+)',
                r'<meta[^>]+name=["\']description["\'][^>]+content=["\']([^"\']+)'):
        m = re.search(pat, page, re.I)
        if m:
            return html.unescape(m.group(1)).strip()
    return ""


def build_news(symbol, name):
    items = []
    base = re.sub(r"\.(NS|BO)$", "", symbol)
    short = re.sub(r"\b(limited|ltd\.?|inc\.?|corporation|corp\.?|company|co\.?)\b", "", name or base, flags=re.I).strip(" ,.")
    queries = [f'"{short}" share' if short else base, f"{base} stock results"]
    jobs = []
    with cf.ThreadPoolExecutor(max_workers=4) as ex:
        for q in queries:
            jobs.append(ex.submit(fetch_google_news, q))
        jobs.append(ex.submit(fetch_yahoo_news, symbol))
        for j in jobs:
            try:
                items += j.result(timeout=20)
            except Exception:
                pass
    seen, uniq = set(), []
    for it in sorted(items, key=lambda x: -x["ts"]):
        key = re.sub(r"\W+", "", it["title"].lower())[:60]
        if not key or key in seen:
            continue
        seen.add(key)
        uniq.append(it)
    uniq = uniq[:20]
    # try to fetch a real description for direct (Yahoo) links
    direct = [it for it in uniq[:10] if "news.google.com" not in it["link"] and not it["summary"]]
    if direct:
        with cf.ThreadPoolExecutor(max_workers=5) as ex:
            for it, desc in zip(direct, ex.map(lambda i: meta_description(i["link"]), direct)):
                it["summary"] = desc
    for it in uniq:
        it["sentiment"], it["score"], it["reasons"], it["category"] = analyse(it["title"], it["summary"])
    return uniq


def news_summary(news):
    pos = sum(1 for n in news if n["sentiment"] == "pos")
    neg = sum(1 for n in news if n["sentiment"] == "neg")
    neu = len(news) - pos - neg
    score = sum(n["score"] for n in news[:12])
    label = "pos" if score >= 3 else "neg" if score <= -3 else "neu"
    return {"pos": pos, "neg": neg, "neu": neu, "score": score, "label": label}


def get_bundle(symbol, force=False):

    with LOCK:
        hit = CACHE.get(key)
        if hit and not force and time.time() - hit[0] < CACHE_TTL:
            return hit[1]
    data = demo_bundle(symbol) if DEMO else live_bundle(symbol)
    with LOCK:

def live_bundle(symbol):
    with cf.ThreadPoolExecutor(max_workers=2) as ex:
        f_chart = ex.submit(fetch_chart, symbol)
        f_sum = ex.submit(fetch_summary, symbol)
        meta, hist = f_chart.result()
        try:
            qs = f_sum.result()
            warn = None
        except Exception as e:
            qs, warn = {}, f"Fundamentals unavailable ({e})"
    fund = flatten_fund(qs)
    name = meta.get("longName") or meta.get("shortName") or fund.get("longName") or symbol
    price = meta.get("regularMarketPrice")
    prev = meta.get("chartPreviousClose") or (hist["c"][-2] if len(hist["c"]) > 1 else price)
    prev = meta.get("previousClose") or prev
    news = build_news(symbol, name)
    return {
        "symbol": symbol, "name": name, "exchange": meta.get("fullExchangeName") or meta.get("exchangeName"),
        "currency": meta.get("currency", ""), "price": price, "prevClose": prev,
        "change": (price - prev) if price and prev else 0,
        "changePct": ((price / prev - 1) * 100) if price and prev else 0,
        "dayHigh": meta.get("regularMarketDayHigh"), "dayLow": meta.get("regularMarketDayLow"),
        "volume": meta.get("regularMarketVolume"),
        "w52High": meta.get("fiftyTwoWeekHigh"), "w52Low": meta.get("fiftyTwoWeekLow"),
        "history": hist, "fund": fund, "earnings": parse_earnings(qs), "profile": parse_profile(qs),
        "news": news, "newsSummary": news_summary(news), "warning": warn,
        "fetchedAt": int(time.time()), "demo": False,
    }

def load_portfolio():
    try:
        with open(PORTFOLIO_FILE, encoding="utf-8") as f:
            p = json.load(f)
    except Exception:
        p = None
    if not isinstance(p, dict):
        p = {"holdings": [], "orders": []}
        if DEMO:
            p["holdings"] = [{"symbol": "RELIANCE.NS", "qty": 40, "avg": 1320}, {"symbol": "TCS.NS", "qty": 25, "avg": 3650},
                             {"symbol": "HDFCBANK.NS", "qty": 120, "avg": 940}, {"symbol": "INFY.NS", "qty": 50, "avg": 1610}]
    p.setdefault("holdings", [])
    p.setdefault("orders", [])
    return p


def save_portfolio(p):
    if DEMO:
        return  # demo mode never overwrites your saved portfolio
    clean = {"holdings": [], "orders": []}
    for h in p.get("holdings", []):
        try:
            clean["holdings"].append({"symbol": str(h["symbol"]).upper()[:24], "qty": float(h.get("qty") or 0),
                                      "avg": float(h.get("avg") or 0)})
        except Exception:
            pass
    for o in p.get("orders", []):
        try:
            clean["orders"].append({"id": str(o["id"])[:40], "symbol": str(o["symbol"]).upper()[:24],
                                    "side": "SELL" if o.get("side") == "SELL" else "BUY",
                                    "type": str(o.get("type", "Limit"))[:20], "qty": float(o.get("qty") or 0),
                                    "price": float(o.get("price") or 0),
                                    "status": o.get("status") if o.get("status") in ("Open", "Executed", "Cancelled") else "Open",
                                    "date": str(o.get("date", ""))[:20]})
        except Exception:
            pass
    tmp = PORTFOLIO_FILE + ".tmp"
    with open(tmp, "w", encoding="utf-8") as f:
        json.dump(clean, f, indent=2)
    os.replace(tmp, PORTFOLIO_FILE)

class Handler(BaseHTTPRequestHandler):
    def log_message(self, *a):
        pass

    def send(self, code, body, ctype="application/json; charset=utf-8"):
        if isinstance(body, (dict, list)):
            body = json.dumps(body, allow_nan=False, default=str).encode()
        elif isinstance(body, str):
            body = body.encode()
        self.send_response(code)
        self.send_header("Content-Type", ctype)
        self.send_header("Content-Length", str(len(body)))
        self.send_header("Cache-Control", "no-store")
        self.end_headers()
        self.wfile.write(body)

    def authorised(self):
        if not PASSWORD:
            return True
        h = self.headers.get("Authorization", "")
        if h.startswith("Basic "):
            try:
                pw = base64.b64decode(h[6:]).decode("utf-8", "ignore").split(":", 1)[-1]
                if hmac.compare_digest(pw, PASSWORD):
                    return True
            except Exception:
                pass
        body = b"Password needed"
        self.send_response(401)
        self.send_header("WWW-Authenticate", 'Basic realm="Stock Desk"')
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)
        return False

    def do_GET(self):
        if not self.authorised():
            return
        u = urllib.parse.urlparse(self.path)
        qs = urllib.parse.parse_qs(u.query)
        try:
            if u.path in ("/", "/index.html"):
                page = INDEX_HTML
                if page is None:
                    with open(os.path.join(HERE, "index.html"), encoding="utf-8") as f:
                        page = f.read()
                page = page.replace("__DEMO__", "true" if DEMO else "false").replace("__HOSTED__", "true" if HOSTED else "false")
                return self.send(200, page, "text/html; charset=utf-8")
            if u.path == "/api/portfolio":
                return self.send(200, load_portfolio())
            if u.path == "/api/search":
                return self.send(200, search(qs.get("q", [""])[0]))
            if u.path == "/api/stock":
                sym = qs.get("symbol", [""])[0]
                if not re.fullmatch(r"[A-Za-z0-9.\-^=&]{1,24}", sym):
                    return self.send(400, {"error": "Bad symbol"})
                return self.send(200, clean_nan(get_bundle(sym, force=qs.get("force") == ["1"])))
            return self.send(404, {"error": "Not found"})
        except Exception as e:
            return self.send(502, {"error": f"{type(e).__name__}: {e}"})

    def do_POST(self):
        if not self.authorised():
            return
        if self.path != "/api/portfolio":
            return self.send(404, {"error": "Not found"})
        try:
            n = int(self.headers.get("Content-Length", 0))
            save_portfolio(json.loads(self.rfile.read(min(n, 2_000_000))))
            return self.send(200, {"ok": True})
        except Exception as e:
            return self.send(400, {"error": str(e)})


def clean_nan(o):
    if isinstance(o, float):
        return None if math.isnan(o) or math.isinf(o) else o
    if isinstance(o, dict):
        return {k: clean_nan(v) for k, v in o.items()}
    if isinstance(o, list):
        return [clean_nan(v) for v in o]
    return o


def main():
    global DEMO, HOSTED
    ap = argparse.ArgumentParser()
    ap.add_argument("--hosted", action="store_true", help="run as an online service (used on Render etc.)")
    ap.add_argument("--port", type=int, default=8765)
    ap.add_argument("--no-browser", action="store_true")
    ap.add_argument("--demo", action="store_true")
    a = ap.parse_args()
    DEMO = a.demo
    HOSTED = a.hosted or bool(os.environ.get("PORT"))
    port = int(os.environ.get("PORT", a.port))
    bind = "0.0.0.0" if HOSTED else "127.0.0.1"
    srv = None
    for p in range(port, port + 20):
        try:
            srv = ThreadingHTTPServer((bind, p), Handler)
            port = p
            break
        except OSError:
            continue
    if not srv:
        sys.exit("Could not find a free port.")
    url = f"http://127.0.0.1:{port}"
    print(f"Stock Desk running at {url}{'  (DEMO DATA)' if DEMO else ''}\nPress Ctrl+C to stop.")
    if not a.no_browser and not HOSTED:
        threading.Timer(0.8, lambda: webbrowser.open(url)).start()
    try:
        srv.serve_forever()
    except KeyboardInterrupt:
        print("\nStopped.")


if __name__ == "__main__":
    main()
